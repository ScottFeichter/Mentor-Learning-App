# The Node Object Model

The NOM is not an official term, but it's a useful way to think about Node's global environment.

Just as the browser has `window` as its global object, Node has `global`. It exposes objects and functions that are available everywhere in a Node program without needing to import them.

Unlike the browser, there is no DOM. Instead, Node's globals give you access to the runtime environment, the process, I/O, timers, and module system.

```js
let global = {

  process: {
    pid: 12345,
    version: "v20.0.0",
    platform: "darwin",
    arch: "arm64",
    env: {
      NODE_ENV: "development",
      PORT: "3000",
      HOME: "/Users/username"
    },
    argv: ["node", "app.js"],
    cwd: function() {},
    exit: function(code) {},
    on: function(event, callback) {},
    stdout: {},
    stderr: {},
    stdin: {}
  },

  console: {
    log: function(message) {},
    error: function(message) {},
    warn: function(message) {},
    table: function(tableData) {},
    time: function(label) {},
    timeEnd: function(label) {}
  },

  Buffer: {
    from: function(data, encoding) {},
    alloc: function(size) {},
    isBuffer: function(obj) {}
  },

  setTimeout: function(callback, ms) {},
  setInterval: function(callback, ms) {},
  clearTimeout: function(id) {},
  clearInterval: function(id) {},
  setImmediate: function(callback) {},
  clearImmediate: function(id) {},

  __filename: "/Users/username/project/app.js",
  __dirname: "/Users/username/project",

  require: function(module) {},

  module: {
    id: ".",
    filename: "/Users/username/project/app.js",
    loaded: true,
    exports: {},
    paths: []
  },

  exports: {},

  URL: function(url, base) {},
  URLSearchParams: function(init) {},

  fetch: function(url, options) {},

  crypto: {
    randomUUID: function() {},
    randomBytes: function(size) {}
  },

  queueMicrotask: function(callback) {},
  structuredClone: function(obj) {},
  atob: function(data) {},
  btoa: function(data) {}

}
```
