# Other Authentication Strategies

Beyond LocalStrategy (username/password), Passport.js supports 500+ authentication strategies. This guide provides an overview of the most common alternatives.

## JWT (JSON Web Tokens)

Stateless token-based authentication, ideal for APIs and SPAs.

### Installation
```bash
npm install passport-jwt jsonwebtoken
```

### Basic Setup
```javascript
const JwtStrategy = require('passport-jwt').Strategy;
const ExtractJwt = require('passport-jwt').ExtractJwt;
const jwt = require('jsonwebtoken');

const opts = {
  jwtFromRequest: ExtractJwt.fromAuthHeaderAsBearerToken(),
  secretOrKey: process.env.JWT_SECRET
};

passport.use(new JwtStrategy(opts, async (payload, done) => {
  try {
    const user = await User.findById(payload.id);
    if (user) {
      return done(null, user);
    }
    return done(null, false);
  } catch (err) {
    return done(err, false);
  }
}));

// Generate token on login
app.post('/login', async (req, res) => {
  const { username, password } = req.body;
  const user = await User.findOne({ username });
  
  if (!user || !await bcrypt.compare(password, user.password)) {
    return res.status(401).json({ message: 'Invalid credentials' });
  }
  
  const token = jwt.sign({ id: user.id }, process.env.JWT_SECRET, {
    expiresIn: '1h'
  });
  
  res.json({ token });
});

// Protected route
app.get('/profile', passport.authenticate('jwt', { session: false }), (req, res) => {
  res.json({ user: req.user });
});
```

**Pros:**
- Stateless (no server-side session)
- Scalable (works across multiple servers)
- Good for APIs and mobile apps

**Cons:**
- Can't invalidate tokens (until expiry)
- Larger payload than session ID
- Vulnerable if token stolen

## OAuth 2.0 (Google, Facebook, GitHub)

Third-party authentication using OAuth providers.

### Google OAuth

```bash
npm install passport-google-oauth20
```

```javascript
const GoogleStrategy = require('passport-google-oauth20').Strategy;

passport.use(new GoogleStrategy({
    clientID: process.env.GOOGLE_CLIENT_ID,
    clientSecret: process.env.GOOGLE_CLIENT_SECRET,
    callbackURL: '/auth/google/callback'
  },
  async (accessToken, refreshToken, profile, done) => {
    try {
      // Find or create user
      let user = await User.findOne({ googleId: profile.id });
      
      if (!user) {
        user = await User.create({
          googleId: profile.id,
          email: profile.emails[0].value,
          name: profile.displayName
        });
      }
      
      return done(null, user);
    } catch (err) {
      return done(err, false);
    }
  }
));

// Routes
app.get('/auth/google',
  passport.authenticate('google', { scope: ['profile', 'email'] })
);

app.get('/auth/google/callback',
  passport.authenticate('google', { failureRedirect: '/login' }),
  (req, res) => {
    res.redirect('/dashboard');
  }
);
```

### GitHub OAuth

```bash
npm install passport-github2
```

```javascript
const GitHubStrategy = require('passport-github2').Strategy;

passport.use(new GitHubStrategy({
    clientID: process.env.GITHUB_CLIENT_ID,
    clientSecret: process.env.GITHUB_CLIENT_SECRET,
    callbackURL: '/auth/github/callback'
  },
  async (accessToken, refreshToken, profile, done) => {
    try {
      let user = await User.findOne({ githubId: profile.id });
      
      if (!user) {
        user = await User.create({
          githubId: profile.id,
          username: profile.username,
          email: profile.emails[0].value
        });
      }
      
      return done(null, user);
    } catch (err) {
      return done(err, false);
    }
  }
));

app.get('/auth/github',
  passport.authenticate('github', { scope: ['user:email'] })
);

app.get('/auth/github/callback',
  passport.authenticate('github', { failureRedirect: '/login' }),
  (req, res) => {
    res.redirect('/dashboard');
  }
);
```

**Pros:**
- No password management
- Trusted third-party verification
- Quick user onboarding

**Cons:**
- Depends on third-party service
- Requires OAuth app setup
- Privacy concerns for users

## API Key Authentication

Simple authentication for APIs using static keys.

```javascript
function authenticateApiKey(req, res, next) {
  const apiKey = req.headers['x-api-key'];
  
  if (!apiKey) {
    return res.status(401).json({ message: 'API key required' });
  }
  
  // Verify API key
  ApiKey.findOne({ key: apiKey }, (err, key) => {
    if (err || !key) {
      return res.status(403).json({ message: 'Invalid API key' });
    }
    
    req.apiKey = key;
    next();
  });
}

app.get('/api/data', authenticateApiKey, (req, res) => {
  res.json({ data: 'Protected data' });
});
```

**Pros:**
- Simple to implement
- Good for server-to-server communication
- No user interaction needed

**Cons:**
- Less secure than other methods
- Hard to rotate keys
- No user context

## Magic Link (Passwordless)

Email-based authentication without passwords.

```javascript
const crypto = require('crypto');

// Request magic link
app.post('/auth/magic-link', async (req, res) => {
  const { email } = req.body;
  
  const user = await User.findOne({ email });
  if (!user) {
    return res.status(404).json({ message: 'User not found' });
  }
  
  // Generate token
  const token = crypto.randomBytes(32).toString('hex');
  
  user.magicLinkToken = token;
  user.magicLinkExpiry = Date.now() + 15 * 60 * 1000; // 15 minutes
  await user.save();
  
  // Send email with link
  const link = `${process.env.BASE_URL}/auth/verify?token=${token}`;
  // sendEmail(email, link);
  
  res.json({ message: 'Magic link sent' });
});

// Verify magic link
app.get('/auth/verify', async (req, res) => {
  const { token } = req.query;
  
  const user = await User.findOne({
    magicLinkToken: token,
    magicLinkExpiry: { $gt: Date.now() }
  });
  
  if (!user) {
    return res.status(400).json({ message: 'Invalid or expired token' });
  }
  
  // Clear token
  user.magicLinkToken = undefined;
  user.magicLinkExpiry = undefined;
  await user.save();
  
  // Log user in
  req.login(user, (err) => {
    if (err) return res.status(500).json({ message: 'Login failed' });
    res.redirect('/dashboard');
  });
});
```

**Pros:**
- No password to remember
- More secure than passwords
- Good user experience

**Cons:**
- Requires email access
- Slower than password login
- Email delivery issues

## Comparison Table

| Strategy | Stateful | Use Case | Complexity | Security |
|----------|----------|----------|------------|----------|
| **Local** | Yes | Traditional web apps | Low | High |
| **JWT** | No | APIs, SPAs, mobile | Medium | Medium |
| **OAuth** | Yes | Social login | Medium | High |
| **API Key** | No | Server-to-server | Low | Low |
| **Magic Link** | Yes | Passwordless | Medium | High |

## Multiple Strategies

You can use multiple strategies in the same app:

```javascript
// Local strategy
passport.use('local', new LocalStrategy(localVerify));

// JWT strategy
passport.use('jwt', new JwtStrategy(jwtOptions, jwtVerify));

// Google OAuth
passport.use('google', new GoogleStrategy(googleOptions, googleVerify));

// Routes
app.post('/login', passport.authenticate('local'), ...);
app.get('/api/data', passport.authenticate('jwt', { session: false }), ...);
app.get('/auth/google', passport.authenticate('google'), ...);
```

## When to Use What

### Use Local Strategy when:
- Building traditional web apps
- Need full control over authentication
- Users expect username/password login
- Session-based authentication is acceptable

### Use JWT when:
- Building RESTful APIs
- Need stateless authentication
- Building mobile apps
- Microservices architecture

### Use OAuth when:
- Want social login
- Don't want to manage passwords
- Need quick user onboarding
- Building consumer-facing apps

### Use API Keys when:
- Server-to-server communication
- Simple authentication needs
- No user interaction required
- Internal APIs

### Use Magic Links when:
- Want passwordless authentication
- Security is important
- Users have reliable email access
- Modern, user-friendly experience

## Key Takeaways

- **LocalStrategy** = Username/password (session-based)
- **JWT** = Token-based (stateless)
- **OAuth** = Third-party authentication
- **API Key** = Simple key-based auth
- **Magic Link** = Email-based passwordless
- Choose based on your app's needs
- Can combine multiple strategies
- Each has trade-offs in security, UX, and complexity
