# Serialize and Deserialize

Serialize and deserialize are two functions in Passport.js that manage how user information is stored in and retrieved from sessions. They're essential for maintaining authentication state across requests.

## The Problem They Solve

Sessions can't store complex objects directly. They need to store simple data (like an ID) and reconstruct the full user object when needed.

```javascript
// Can't do this - too much data in session
req.session.user = {
  id: 123,
  username: 'john',
  email: 'john@example.com',
  profile: { ... },
  settings: { ... }
  // ... lots more data
};

// Instead, store just the ID
req.session.userId = 123;
// Then retrieve full user when needed
```

## What They Do

### serializeUser
Converts a user object into a simple identifier (usually ID) to store in the session.

**When it runs:** After successful authentication

```javascript
passport.serializeUser((user, done) => {
  done(null, user.id); // Store only the ID
});
```

### deserializeUser
Converts the stored identifier back into a full user object.

**When it runs:** On every subsequent request (if session exists)

```javascript
passport.deserializeUser(async (id, done) => {
  try {
    const user = await User.findById(id);
    done(null, user); // Attach full user to req.user
  } catch (err) {
    done(err);
  }
});
```

## Complete Flow

```
Login Request
    ↓
1. User submits credentials
    ↓
2. Strategy verifies credentials
    ↓
3. Strategy calls done(null, user) with full user object
    ↓
4. serializeUser(user, done) called
    ↓
5. Returns user.id → stored in session
    ↓
6. Session cookie sent to browser

Subsequent Request
    ↓
1. Browser sends session cookie
    ↓
2. Express retrieves session data (contains user.id)
    ↓
3. deserializeUser(id, done) called
    ↓
4. Retrieves full user from database
    ↓
5. Attaches user to req.user
    ↓
6. Route handler has access to req.user
```

## Basic Example

```javascript
const passport = require('passport');
const User = require('./models/User');

// Serialize: user object → user ID
passport.serializeUser((user, done) => {
  console.log('Serializing user:', user.id);
  done(null, user.id);
});

// Deserialize: user ID → user object
passport.deserializeUser(async (id, done) => {
  console.log('Deserializing user:', id);
  try {
    const user = await User.findById(id);
    done(null, user);
  } catch (err) {
    done(err);
  }
});
```

## What Gets Stored

### In Session (After Serialize)
```javascript
{
  passport: {
    user: 123  // Just the ID
  }
}
```

### In req.user (After Deserialize)
```javascript
{
  id: 123,
  username: 'john',
  email: 'john@example.com',
  createdAt: '2025-01-15',
  // ... full user object
}
```

## Common Patterns

### Store Just ID (Most Common)
```javascript
passport.serializeUser((user, done) => {
  done(null, user.id);
});

passport.deserializeUser(async (id, done) => {
  const user = await User.findById(id);
  done(null, user);
});
```

### Store Multiple Fields
```javascript
passport.serializeUser((user, done) => {
  done(null, { id: user.id, role: user.role });
});

passport.deserializeUser(async (data, done) => {
  const user = await User.findById(data.id);
  user.role = data.role; // Add cached role
  done(null, user);
});
```

### Store Entire User (Not Recommended)
```javascript
// Don't do this - wastes session storage
passport.serializeUser((user, done) => {
  done(null, user); // Stores entire object
});

passport.deserializeUser((user, done) => {
  done(null, user); // Just returns it
});
```

### Selective User Data
```javascript
passport.deserializeUser(async (id, done) => {
  const user = await User.findById(id).select('username email role');
  done(null, user); // Only load needed fields
});
```

## Error Handling

```javascript
passport.deserializeUser(async (id, done) => {
  try {
    const user = await User.findById(id);
    
    if (!user) {
      // User deleted after login
      return done(null, false);
    }
    
    done(null, user);
  } catch (err) {
    done(err);
  }
});
```

## Performance Considerations

### Problem: Database Query on Every Request
```javascript
// This runs on EVERY request
passport.deserializeUser(async (id, done) => {
  const user = await User.findById(id); // Database hit
  done(null, user);
});
```

### Solution 1: Cache User Data
```javascript
const cache = new Map();

passport.deserializeUser(async (id, done) => {
  // Check cache first
  if (cache.has(id)) {
    return done(null, cache.get(id));
  }
  
  const user = await User.findById(id);
  cache.set(id, user);
  done(null, user);
});
```

### Solution 2: Store More in Session
```javascript
passport.serializeUser((user, done) => {
  done(null, {
    id: user.id,
    username: user.username,
    role: user.role
  });
});

passport.deserializeUser((userData, done) => {
  // No database query needed for basic info
  done(null, userData);
});
```

### Solution 3: Lazy Loading
```javascript
passport.deserializeUser((id, done) => {
  // Return minimal object with ID
  done(null, { id });
});

// Load full user only when needed
async function loadFullUser(req, res, next) {
  if (req.user && !req.user.username) {
    req.user = await User.findById(req.user.id);
  }
  next();
}

app.get('/profile', loadFullUser, (req, res) => {
  res.json(req.user); // Full user loaded
});
```

## Common Mistakes

### Mistake 1: Not Calling done()
```javascript
// Wrong
passport.serializeUser((user, done) => {
  user.id; // Forgot to call done()
});

// Correct
passport.serializeUser((user, done) => {
  done(null, user.id);
});
```

### Mistake 2: Serializing Entire User
```javascript
// Wrong - wastes session storage
passport.serializeUser((user, done) => {
  done(null, user);
});

// Correct - store only ID
passport.serializeUser((user, done) => {
  done(null, user.id);
});
```

### Mistake 3: Not Handling Errors
```javascript
// Wrong - no error handling
passport.deserializeUser(async (id, done) => {
  const user = await User.findById(id);
  done(null, user);
});

// Correct
passport.deserializeUser(async (id, done) => {
  try {
    const user = await User.findById(id);
    done(null, user);
  } catch (err) {
    done(err);
  }
});
```

### Mistake 4: Forgetting to Configure
```javascript
// Wrong - forgot to set up serialize/deserialize
passport.use(new LocalStrategy(verify));
// Authentication will fail!

// Correct
passport.use(new LocalStrategy(verify));
passport.serializeUser((user, done) => done(null, user.id));
passport.deserializeUser(async (id, done) => {
  const user = await User.findById(id);
  done(null, user);
});
```

## Debugging

```javascript
passport.serializeUser((user, done) => {
  console.log('SERIALIZE - Storing user ID:', user.id);
  done(null, user.id);
});

passport.deserializeUser(async (id, done) => {
  console.log('DESERIALIZE - Loading user ID:', id);
  try {
    const user = await User.findById(id);
    console.log('DESERIALIZE - User loaded:', user ? user.username : 'not found');
    done(null, user);
  } catch (err) {
    console.error('DESERIALIZE - Error:', err);
    done(err);
  }
});
```

## With Different Databases

### MongoDB (Mongoose)
```javascript
passport.serializeUser((user, done) => {
  done(null, user._id);
});

passport.deserializeUser(async (id, done) => {
  const user = await User.findById(id);
  done(null, user);
});
```

### PostgreSQL (Sequelize)
```javascript
passport.serializeUser((user, done) => {
  done(null, user.id);
});

passport.deserializeUser(async (id, done) => {
  const user = await User.findByPk(id);
  done(null, user);
});
```

### In-Memory (Array)
```javascript
const users = [
  { id: 1, username: 'john' },
  { id: 2, username: 'jane' }
];

passport.serializeUser((user, done) => {
  done(null, user.id);
});

passport.deserializeUser((id, done) => {
  const user = users.find(u => u.id === id);
  done(null, user);
});
```

## Complete Working Example

```javascript
const passport = require('passport');
const LocalStrategy = require('passport-local').Strategy;
const User = require('./models/User');

// Configure strategy
passport.use(new LocalStrategy(
  async (username, password, done) => {
    try {
      const user = await User.findOne({ username });
      if (!user || !await user.validPassword(password)) {
        return done(null, false);
      }
      return done(null, user); // Full user object
    } catch (err) {
      return done(err);
    }
  }
));

// Serialize: Store only ID in session
passport.serializeUser((user, done) => {
  done(null, user.id);
});

// Deserialize: Retrieve full user from ID
passport.deserializeUser(async (id, done) => {
  try {
    const user = await User.findById(id);
    done(null, user); // Attached to req.user
  } catch (err) {
    done(err);
  }
});

// Routes
app.post('/login', passport.authenticate('local'), (req, res) => {
  // serializeUser was called, session created
  res.json({ user: req.user });
});

app.get('/profile', (req, res) => {
  // deserializeUser was called, req.user populated
  if (!req.isAuthenticated()) {
    return res.status(401).send('Not authenticated');
  }
  res.json({ user: req.user });
});
```

## Key Takeaways

- **serializeUser** = User object → ID (store in session)
- **deserializeUser** = ID → User object (attach to req.user)
- **serializeUser runs:** Once after successful login
- **deserializeUser runs:** On every request with a session
- **Store minimal data:** Usually just user ID
- **Handle errors:** Always use try/catch
- **Performance:** Consider caching to reduce database queries
- **Required:** Both must be configured for Passport to work
