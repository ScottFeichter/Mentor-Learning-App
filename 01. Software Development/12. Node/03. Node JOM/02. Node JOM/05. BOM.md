# The Browser Object Model

&nbsp;

The **Browser Object Model (BOM)** refers to properties that live in <span class="purple-bold">globalThis</span>.

These **BOM** properties represent the browser itself.

There is no official standard for the **BOM** (unlike the DOM).

But all major browsers implement the same core objects.

## Programatic Access To Browser Components

In this way <span class="purple-bold">globalThis</span> gives access to such browser components as:

- the URL
- navigation history
- screen info
- browser storage

Notice this also include the DOM - the programatic representation of the HTML/CSS as a treeish structure.

```js
let globalThis = {

  window: globalThis;

    // ... etc JS built-ins

  navigator: {
    userAgent: "Mozilla/5.0 ...",
    language: "en-US",
    onLine: true,
    geolocation: {},
    clipboard: {},
    permissions: {}
  },

  location: {
    href: "https://example.com/page?q=hello#section",
    hostname: "example.com",
    pathname: "/page",
    search: "?q=hello",
    hash: "#section",
    reload: function() {},
    assign: function(url) {},
    replace: function(url) {}
  },

  history: {
    length: 5,
    back: function() {},
    forward: function() {},
    go: function(n) {},
    pushState: function(state, title, url) {},
    replaceState: function(state, title, url) {}
  },

  screen: {
    userScreenInfo: {
      width: 1920,
      height: 1080,
      orientation: 'landscape',
      colorDepth: 24
    }
  },

  localStorage: {
    getItem: function(key) {
      return this[key];
    },
    setItem: function(key, value) {
      this[key] = value;
    }
  },

  sessionStorage: {
    getItem: function(key) {
      return this[key];
    },

    setItem: function(key, value) {
      this[key] = value;
    }
  },

  console: {
    log: function(message) {
      // Implementation for logging messages
    },
    table: function(tableData) {
        // Implementation for displaying tabular data
    },
  },

  document: {
    title: "My Page",
    URL: "https://example.com",
    cookie: "",

    documentElement: {
      tagName: "HTML",

      head: {
        tagName: "HEAD",
        children: [
          { tagName: "META", charset: "UTF-8" },
          { tagName: "META", name: "viewport", content: "width=device-width" },
          { tagName: "TITLE", textContent: "My Page" },
          { tagName: "LINK", rel: "stylesheet", href: "styles.css" },
          { tagName: "SCRIPT", src: "app.js" },
          { tagName: "STYLE", textContent: "body { margin: 0; }" }
        ]
      },

      body: {
        tagName: "BODY",
        children: [
          { tagName: "DIV", id: "root", children: [
            { tagName: "H1", textContent: "Hello World" },
            { tagName: "P", textContent: "Some paragraph text" },
            { tagName: "IMG", src: "photo.png", alt: "A photo" },
            { tagName: "FORM", children: [
              { tagName: "INPUT", type: "text", name: "username" },
              { tagName: "BUTTON", textContent: "Submit" }
            ]}
          ]}
        ]
      }
    },

    forms: [],
    images: [],
    links: []
  },

  fetch: function(url, options) {},
  XMLHttpRequest: function() {},

  setTimeout: function(callback, ms) {},
  setInterval: function(callback, ms) {},
  clearTimeout: function(id) {},
  clearInterval: function(id) {},

  alert: function(message) {},
  confirm: function(message) {},
  prompt: function(message) {}

}
```
