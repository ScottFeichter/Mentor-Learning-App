# BFS Path

Breadth-First Search to find a path through a matrix.

```javascript
function bfsPath(matrix, startNode, endValue) {
  const q = [startNode];
  const v = new Set().add(`${startNode[0]},${startNode[1]}`);
  const res = [];

  while (q.length) {
    // 1. get currNode
    const currNode = q.shift();

    // 2. do some things
    res.push(currNode);

    if (matrix[currNode[0]][currNode[1]] === endValue) {
      return res;
    }

    // 3. find neighbors
    const neighbors = findNeighbors(currNode, matrix);
    neighbors.forEach(el => {
      const strEl = `${el[0]},${el[1]}`;

      if (!v.has(strEl)) {
        v.add(strEl);
        q.push(el);
      }
    });
  }

  return false;
}
```
