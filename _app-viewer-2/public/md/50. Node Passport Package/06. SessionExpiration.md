# Session Expiration

Session expiration is the automatic termination of a user's authenticated session after a specified period of time or inactivity. It's a critical security feature that prevents unauthorized access if a user forgets to log out or leaves their device unattended.

## Why Session Expiration Matters

**Security Reasons:**
- Prevents unauthorized access on shared/public computers
- Limits exposure if session ID is stolen
- Reduces risk of session hijacking
- Protects against forgotten logout

**Resource Management:**
- Frees up server memory/storage
- Prevents accumulation of abandoned sessions
- Reduces database/cache load

## Types of Session Expiration

### 1. Absolute Expiration (Fixed Timeout)
Session expires after a fixed time from creation, regardless of activity.

```javascript
app.use(session({
  secret: 'secret',
  cookie: {
    maxAge: 30 * 60 * 1000  // 30 minutes from login
  }
}));
```

**Use case:** High-security applications (banking, admin panels)

### 2. Idle Expiration (Inactivity Timeout)
Session expires after a period of inactivity. Each request resets the timer.

```javascript
app.use(session({
  secret: 'secret',
  rolling: true,  // Reset expiration on every request
  cookie: {
    maxAge: 15 * 60 * 1000  // 15 minutes of inactivity
  }
}));
```

**Use case:** Most web applications (social media, e-commerce)

### 3. Hybrid Approach
Combines both absolute and idle expiration.

```javascript
app.use(session({
  secret: 'secret',
  rolling: true,
  cookie: {
    maxAge: 15 * 60 * 1000  // 15 min idle timeout
  }
}));

// Middleware to enforce absolute timeout
app.use((req, res, next) => {
  if (req.session.createdAt) {
    const absoluteTimeout = 2 * 60 * 60 * 1000; // 2 hours max
    const sessionAge = Date.now() - req.session.createdAt;
    
    if (sessionAge > absoluteTimeout) {
      return req.session.destroy(() => {
        res.status(401).json({ message: 'Session expired' });
      });
    }
  } else {
    req.session.createdAt = Date.now();
  }
  next();
});
```

**Use case:** Enterprise applications requiring both security and usability

## Configuration Options

### Basic Configuration
```javascript
const session = require('express-session');

app.use(session({
  secret: 'your-secret-key',
  resave: false,
  saveUninitialized: false,
  rolling: true,              // Reset expiration on activity
  cookie: {
    maxAge: 1000 * 60 * 30,   // 30 minutes
    httpOnly: true,
    secure: true,
    sameSite: 'strict'
  }
}));
```

### Key Options Explained

- **maxAge** - Time in milliseconds until cookie expires
- **rolling: true** - Resets maxAge on every request (idle timeout)
- **rolling: false** - Cookie expires at fixed time (absolute timeout)
- **resave: false** - Don't save session if unmodified (performance)
- **saveUninitialized: false** - Don't save empty sessions

## Implementation Examples

### Example 1: Simple Idle Timeout
```javascript
app.use(session({
  secret: 'secret',
  rolling: true,
  cookie: { maxAge: 20 * 60 * 1000 }  // 20 min idle
}));

app.get('/dashboard', (req, res) => {
  if (!req.session.userId) {
    return res.status(401).json({ message: 'Session expired, please login' });
  }
  res.json({ message: 'Dashboard data' });
});
```

### Example 2: Custom Expiration Logic
```javascript
// Store login time in session
app.post('/login', (req, res) => {
  // Verify credentials...
  req.session.userId = user.id;
  req.session.loginTime = Date.now();
  res.json({ message: 'Logged in' });
});

// Middleware to check session expiration
function checkSessionExpiration(req, res, next) {
  if (!req.session.userId) {
    return res.status(401).json({ message: 'Not authenticated' });
  }
  
  const sessionDuration = Date.now() - req.session.loginTime;
  const maxDuration = 60 * 60 * 1000; // 1 hour
  
  if (sessionDuration > maxDuration) {
    req.session.destroy();
    return res.status(401).json({ message: 'Session expired' });
  }
  
  next();
}

app.get('/profile', checkSessionExpiration, (req, res) => {
  res.json({ user: req.session.userId });
});
```

### Example 3: Warning Before Expiration
```javascript
app.get('/session-status', (req, res) => {
  if (!req.session.userId) {
    return res.json({ authenticated: false });
  }
  
  const loginTime = req.session.loginTime;
  const maxAge = 30 * 60 * 1000; // 30 minutes
  const elapsed = Date.now() - loginTime;
  const remaining = maxAge - elapsed;
  
  res.json({
    authenticated: true,
    remainingTime: remaining,
    willExpireIn: Math.floor(remaining / 1000) + ' seconds'
  });
});
```

### Example 4: Extend Session on Activity
```javascript
app.use(session({
  secret: 'secret',
  rolling: false,  // Don't auto-extend
  cookie: { maxAge: 30 * 60 * 1000 }
}));

// Manual session extension endpoint
app.post('/extend-session', (req, res) => {
  if (!req.session.userId) {
    return res.status(401).json({ message: 'No active session' });
  }
  
  // Reset session expiration
  req.session.cookie.maxAge = 30 * 60 * 1000;
  res.json({ message: 'Session extended' });
});
```

## Client-Side Handling

### Detect Expiration and Redirect
```javascript
// Frontend code
async function fetchProtectedData() {
  try {
    const response = await fetch('/api/profile');
    
    if (response.status === 401) {
      // Session expired
      alert('Your session has expired. Please login again.');
      window.location.href = '/login';
      return;
    }
    
    const data = await response.json();
    return data;
  } catch (error) {
    console.error('Error:', error);
  }
}
```

### Session Timeout Warning
```javascript
// Frontend: Warn user before expiration
let warningTimeout;
let logoutTimeout;

function resetTimers() {
  clearTimeout(warningTimeout);
  clearTimeout(logoutTimeout);
  
  // Warn 2 minutes before expiration
  warningTimeout = setTimeout(() => {
    alert('Your session will expire in 2 minutes');
  }, 28 * 60 * 1000);
  
  // Auto logout after 30 minutes
  logoutTimeout = setTimeout(() => {
    window.location.href = '/logout';
  }, 30 * 60 * 1000);
}

// Reset timers on user activity
document.addEventListener('click', resetTimers);
document.addEventListener('keypress', resetTimers);

resetTimers(); // Initialize
```

## Common Expiration Times

| Application Type | Typical Timeout | Type |
|-----------------|-----------------|------|
| Banking/Financial | 5-15 minutes | Absolute |
| Admin Panels | 15-30 minutes | Idle |
| E-commerce | 30-60 minutes | Idle |
| Social Media | 1-7 days | Idle |
| Email Services | 1-30 days | Idle |
| "Remember Me" | 30-90 days | Absolute |

## Best Practices

1. **Match timeout to security needs**
   - High security = shorter timeout
   - User convenience = longer timeout

2. **Use idle timeout for most apps**
   - Better user experience
   - Automatically extends with activity

3. **Warn users before expiration**
   - Give option to extend session
   - Prevent data loss

4. **Implement "Remember Me" separately**
   - Use longer-lived token
   - Store in separate cookie

5. **Clear sensitive data on expiration**
   - Destroy session completely
   - Clear cookies

6. **Log session expirations**
   - Track for security monitoring
   - Identify unusual patterns

## Session Expiration with Passport.js

```javascript
const passport = require('passport');

app.use(session({
  secret: 'secret',
  rolling: true,
  cookie: { maxAge: 30 * 60 * 1000 }
}));

app.use(passport.initialize());
app.use(passport.session());

// Middleware to check expiration
function ensureAuthenticated(req, res, next) {
  if (req.isAuthenticated()) {
    return next();
  }
  res.status(401).json({ 
    message: 'Session expired or not authenticated',
    redirect: '/login'
  });
}

app.get('/dashboard', ensureAuthenticated, (req, res) => {
  res.json({ user: req.user });
});
```

## Handling Expired Sessions

```javascript
// Global error handler for expired sessions
app.use((err, req, res, next) => {
  if (err.message === 'Session expired') {
    return res.status(401).json({
      error: 'Session expired',
      message: 'Please login again',
      redirect: '/login'
    });
  }
  next(err);
});

// Middleware to check and handle expiration
app.use((req, res, next) => {
  if (req.session && req.session.userId) {
    // Check if session is still valid
    if (req.session.cookie.expires < Date.now()) {
      req.session.destroy();
      return res.status(401).json({ message: 'Session expired' });
    }
  }
  next();
});
```

## Key Takeaways

- **Session expiration** = Automatic logout after time period
- **Idle timeout** = Expires after inactivity (rolling: true)
- **Absolute timeout** = Expires after fixed time (rolling: false)
- **Security vs Usability** = Balance timeout length appropriately
- **Always warn users** = Prevent unexpected logouts and data loss
- **Clear sessions properly** = Destroy session and clear cookies
- **Match timeout to risk** = Sensitive apps need shorter timeouts
