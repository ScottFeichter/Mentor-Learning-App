# Introduction to Passport.js

Passport is authentication middleware for Node.js. It simplifies the process of implementing authentication by providing a consistent API and supporting over 500 authentication strategies.

## What is Passport.js?

Passport is a middleware that:
- Handles authentication logic
- Supports multiple authentication strategies (local, OAuth, JWT, etc.)
- Integrates with Express sessions
- Provides a consistent API regardless of strategy
- Is modular and unobtrusive

## Why Use Passport?

**Without Passport:**
```javascript
app.post('/login', async (req, res) => {
  const { username, password } = req.body;
  const user = await User.findOne({ username });
  
  if (!user) return res.status(401).send('Invalid credentials');
  
  const isValid = await bcrypt.compare(password, user.password);
  if (!isValid) return res.status(401).send('Invalid credentials');
  
  req.session.userId = user.id;
  req.session.username = user.username;
  // ... more session setup
  
  res.json({ message: 'Logged in' });
});

// Repeat similar logic for OAuth, JWT, etc.
```

**With Passport:**
```javascript
app.post('/login', passport.authenticate('local'), (req, res) => {
  res.json({ message: 'Logged in', user: req.user });
});

// Same pattern works for any strategy
app.get('/auth/google', passport.authenticate('google'));
```

## Core Concepts

### 1. Strategies
A strategy is a method of authentication (username/password, Google OAuth, JWT, etc.).

```javascript
const LocalStrategy = require('passport-local').Strategy;

passport.use(new LocalStrategy((username, password, done) => {
  // Verify credentials
  // Return user or false
}));
```

### 2. Middleware
Passport provides middleware to authenticate requests.

```javascript
app.post('/login', passport.authenticate('local'), (req, res) => {
  // If this runs, authentication succeeded
});
```

### 3. Sessions
Passport integrates with Express sessions to persist authentication.

```javascript
passport.serializeUser((user, done) => {
  done(null, user.id); // Store user ID in session
});

passport.deserializeUser((id, done) => {
  User.findById(id, (err, user) => {
    done(err, user); // Retrieve user from ID
  });
});
```

## Basic Setup

### Installation

```bash
npm install passport passport-local express-session bcrypt
```

### Configuration

```javascript
const express = require('express');
const session = require('express-session');
const passport = require('passport');
const LocalStrategy = require('passport-local').Strategy;

const app = express();

// 1. Body parser
app.use(express.json());
app.use(express.urlencoded({ extended: false }));

// 2. Session middleware (must come before passport)
app.use(session({
  secret: 'your-secret-key',
  resave: false,
  saveUninitialized: false,
  cookie: {
    httpOnly: true,
    secure: false, // true in production
    maxAge: 24 * 60 * 60 * 1000
  }
}));

// 3. Initialize Passport
app.use(passport.initialize());
app.use(passport.session());

// 4. Configure strategy
passport.use(new LocalStrategy(
  async (username, password, done) => {
    try {
      const user = await User.findOne({ username });
      if (!user) return done(null, false, { message: 'User not found' });
      
      const isValid = await bcrypt.compare(password, user.password);
      if (!isValid) return done(null, false, { message: 'Incorrect password' });
      
      return done(null, user);
    } catch (err) {
      return done(err);
    }
  }
));

// 5. Serialize/deserialize
passport.serializeUser((user, done) => {
  done(null, user.id);
});

passport.deserializeUser(async (id, done) => {
  try {
    const user = await User.findById(id);
    done(null, user);
  } catch (err) {
    done(err);
  }
});

// 6. Routes
app.post('/login', passport.authenticate('local'), (req, res) => {
  res.json({ message: 'Logged in', user: req.user });
});

app.get('/profile', (req, res) => {
  if (!req.isAuthenticated()) {
    return res.status(401).send('Not authenticated');
  }
  res.json({ user: req.user });
});

app.post('/logout', (req, res) => {
  req.logout((err) => {
    if (err) return res.status(500).send('Error');
    res.send('Logged out');
  });
});

app.listen(3000);
```

## Key Methods

### passport.authenticate()
Authenticates a request using a strategy.

```javascript
// Redirect on success/failure
app.post('/login', 
  passport.authenticate('local', {
    successRedirect: '/dashboard',
    failureRedirect: '/login'
  })
);

// Custom callback
app.post('/login', (req, res, next) => {
  passport.authenticate('local', (err, user, info) => {
    if (err) return next(err);
    if (!user) return res.status(401).json({ message: info.message });
    
    req.login(user, (err) => {
      if (err) return next(err);
      return res.json({ user });
    });
  })(req, res, next);
});
```

### req.login()
Manually establish a login session.

```javascript
req.login(user, (err) => {
  if (err) return next(err);
  res.json({ message: 'Logged in' });
});
```

### req.logout()
Terminate a login session.

```javascript
req.logout((err) => {
  if (err) return next(err);
  res.redirect('/');
});
```

### req.isAuthenticated()
Check if user is authenticated.

```javascript
if (req.isAuthenticated()) {
  // User is logged in
}
```

### req.user
Access the authenticated user.

```javascript
app.get('/profile', (req, res) => {
  res.json({ username: req.user.username });
});
```

## Authentication Flow

```
1. User submits credentials
   ↓
2. passport.authenticate() called
   ↓
3. Strategy verify function executes
   ↓
4. If valid, done(null, user) called
   ↓
5. passport.serializeUser() stores user.id in session
   ↓
6. Session cookie sent to browser
   ↓
7. Subsequent requests include cookie
   ↓
8. passport.deserializeUser() retrieves user from ID
   ↓
9. req.user populated with user object
```

## Common Strategies

### Local Strategy (Username/Password)
```javascript
const LocalStrategy = require('passport-local').Strategy;
passport.use(new LocalStrategy(verifyFunction));
```

### Google OAuth
```javascript
const GoogleStrategy = require('passport-google-oauth20').Strategy;
passport.use(new GoogleStrategy(config, verifyFunction));
```

### JWT (Token-Based)
```javascript
const JwtStrategy = require('passport-jwt').Strategy;
passport.use(new JwtStrategy(config, verifyFunction));
```

### GitHub OAuth
```javascript
const GitHubStrategy = require('passport-github2').Strategy;
passport.use(new GitHubStrategy(config, verifyFunction));
```

## Protecting Routes

### Simple Check
```javascript
app.get('/dashboard', (req, res) => {
  if (!req.isAuthenticated()) {
    return res.status(401).send('Unauthorized');
  }
  res.send('Dashboard');
});
```

### Middleware
```javascript
function ensureAuthenticated(req, res, next) {
  if (req.isAuthenticated()) {
    return next();
  }
  res.status(401).json({ message: 'Authentication required' });
}

app.get('/dashboard', ensureAuthenticated, (req, res) => {
  res.send('Dashboard');
});
```

## Advantages of Passport

✅ **Modular** - Use only what you need
✅ **Flexible** - 500+ strategies available
✅ **Consistent API** - Same pattern for all strategies
✅ **Well-maintained** - Active community
✅ **Express integration** - Works seamlessly with Express
✅ **Session support** - Built-in session management
✅ **Customizable** - Full control over authentication logic

## When to Use Passport

**Use Passport when:**
- Building traditional web apps with sessions
- Need multiple authentication methods
- Want a proven, battle-tested solution
- Need OAuth integration
- Building with Express

**Consider alternatives when:**
- Building stateless APIs (consider JWT directly)
- Very simple authentication needs
- Using a different framework
- Need maximum control over every detail

## Common Patterns

### Multiple Strategies
```javascript
passport.use('local-login', new LocalStrategy(loginVerify));
passport.use('local-signup', new LocalStrategy(signupVerify));

app.post('/login', passport.authenticate('local-login'));
app.post('/signup', passport.authenticate('local-signup'));
```

### Custom Callback
```javascript
app.post('/login', (req, res, next) => {
  passport.authenticate('local', (err, user, info) => {
    if (err) return next(err);
    if (!user) return res.status(401).json(info);
    
    req.login(user, (err) => {
      if (err) return next(err);
      return res.json({ user });
    });
  })(req, res, next);
});
```

### Flash Messages
```javascript
const flash = require('connect-flash');
app.use(flash());

app.post('/login',
  passport.authenticate('local', {
    successRedirect: '/dashboard',
    failureRedirect: '/login',
    failureFlash: true
  })
);
```

## Key Takeaways

- **Passport** = Authentication middleware for Node.js
- **Strategy** = Method of authentication (local, OAuth, JWT)
- **Serialize/Deserialize** = Convert user to/from session
- **req.user** = Authenticated user object
- **req.isAuthenticated()** = Check if logged in
- Passport handles the complexity of authentication
- Works seamlessly with Express sessions
- Supports 500+ authentication strategies
