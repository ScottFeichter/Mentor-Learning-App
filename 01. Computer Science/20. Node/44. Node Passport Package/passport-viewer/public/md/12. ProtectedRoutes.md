# Protected Routes

Protected routes are endpoints that require authentication before access. They ensure only logged-in users can access certain resources.

## Basic Protection

### Simple Check
```javascript
app.get('/profile', (req, res) => {
  if (!req.isAuthenticated()) {
    return res.status(401).json({ message: 'Authentication required' });
  }
  
  res.json({ user: req.user });
});
```

### Middleware Function
```javascript
function ensureAuthenticated(req, res, next) {
  if (req.isAuthenticated()) {
    return next();
  }
  res.status(401).json({ message: 'Authentication required' });
}

app.get('/profile', ensureAuthenticated, (req, res) => {
  res.json({ user: req.user });
});

app.get('/dashboard', ensureAuthenticated, (req, res) => {
  res.send('Dashboard');
});
```

## Middleware Patterns

### Redirect to Login
```javascript
function ensureAuthenticated(req, res, next) {
  if (req.isAuthenticated()) {
    return next();
  }
  res.redirect('/login');
}
```

### JSON Response
```javascript
function ensureAuthenticated(req, res, next) {
  if (req.isAuthenticated()) {
    return next();
  }
  res.status(401).json({ 
    message: 'Authentication required',
    redirect: '/login'
  });
}
```

### With Flash Messages
```javascript
function ensureAuthenticated(req, res, next) {
  if (req.isAuthenticated()) {
    return next();
  }
  req.flash('error', 'Please log in to access this page');
  res.redirect('/login');
}
```

## Role-Based Access Control (RBAC)

### Check User Role
```javascript
function requireRole(role) {
  return (req, res, next) => {
    if (!req.isAuthenticated()) {
      return res.status(401).json({ message: 'Authentication required' });
    }
    
    if (req.user.role !== role) {
      return res.status(403).json({ message: 'Insufficient permissions' });
    }
    
    next();
  };
}

// Usage
app.get('/admin', requireRole('admin'), (req, res) => {
  res.send('Admin panel');
});

app.get('/moderator', requireRole('moderator'), (req, res) => {
  res.send('Moderator panel');
});
```

### Check Multiple Roles
```javascript
function requireAnyRole(...roles) {
  return (req, res, next) => {
    if (!req.isAuthenticated()) {
      return res.status(401).json({ message: 'Authentication required' });
    }
    
    if (!roles.includes(req.user.role)) {
      return res.status(403).json({ message: 'Insufficient permissions' });
    }
    
    next();
  };
}

// Usage
app.get('/staff', requireAnyRole('admin', 'moderator', 'staff'), (req, res) => {
  res.send('Staff area');
});
```

### Check Permissions
```javascript
function requirePermission(permission) {
  return (req, res, next) => {
    if (!req.isAuthenticated()) {
      return res.status(401).json({ message: 'Authentication required' });
    }
    
    if (!req.user.permissions.includes(permission)) {
      return res.status(403).json({ message: 'Insufficient permissions' });
    }
    
    next();
  };
}

// Usage
app.delete('/posts/:id', requirePermission('delete_posts'), (req, res) => {
  // Delete post
});

app.post('/users', requirePermission('create_users'), (req, res) => {
  // Create user
});
```

## Resource Ownership

### Check Resource Owner
```javascript
async function requireOwnership(req, res, next) {
  if (!req.isAuthenticated()) {
    return res.status(401).json({ message: 'Authentication required' });
  }
  
  const post = await Post.findById(req.params.id);
  
  if (!post) {
    return res.status(404).json({ message: 'Post not found' });
  }
  
  if (post.userId.toString() !== req.user.id.toString()) {
    return res.status(403).json({ message: 'Not authorized' });
  }
  
  req.post = post; // Attach to request
  next();
}

// Usage
app.put('/posts/:id', requireOwnership, (req, res) => {
  // Update req.post
  res.json({ message: 'Post updated' });
});

app.delete('/posts/:id', requireOwnership, (req, res) => {
  // Delete req.post
  res.json({ message: 'Post deleted' });
});
```

### Owner or Admin
```javascript
async function requireOwnerOrAdmin(req, res, next) {
  if (!req.isAuthenticated()) {
    return res.status(401).json({ message: 'Authentication required' });
  }
  
  const post = await Post.findById(req.params.id);
  
  if (!post) {
    return res.status(404).json({ message: 'Post not found' });
  }
  
  const isOwner = post.userId.toString() === req.user.id.toString();
  const isAdmin = req.user.role === 'admin';
  
  if (!isOwner && !isAdmin) {
    return res.status(403).json({ message: 'Not authorized' });
  }
  
  req.post = post;
  next();
}
```

## Combining Middleware

### Multiple Checks
```javascript
app.delete('/posts/:id',
  ensureAuthenticated,
  requireRole('admin'),
  async (req, res) => {
    // Delete post
  }
);
```

### Reusable Chains
```javascript
const adminOnly = [ensureAuthenticated, requireRole('admin')];
const staffOnly = [ensureAuthenticated, requireAnyRole('admin', 'staff')];

app.get('/admin/users', adminOnly, (req, res) => {
  // Admin only
});

app.get('/staff/reports', staffOnly, (req, res) => {
  // Staff only
});
```

## Protecting Multiple Routes

### Router-Level Middleware
```javascript
const express = require('express');
const router = express.Router();

// Apply to all routes in this router
router.use(ensureAuthenticated);

router.get('/profile', (req, res) => {
  res.json({ user: req.user });
});

router.get('/settings', (req, res) => {
  res.send('Settings');
});

router.get('/dashboard', (req, res) => {
  res.send('Dashboard');
});

module.exports = router;
```

### Protect Entire Path
```javascript
// Protect all /api routes
app.use('/api', ensureAuthenticated);

app.get('/api/users', (req, res) => {
  // Protected
});

app.get('/api/posts', (req, res) => {
  // Protected
});

// Public routes
app.get('/login', (req, res) => {
  // Not protected
});
```

## API Token Authentication

### Bearer Token
```javascript
function authenticateToken(req, res, next) {
  const authHeader = req.headers['authorization'];
  const token = authHeader && authHeader.split(' ')[1];
  
  if (!token) {
    return res.status(401).json({ message: 'Token required' });
  }
  
  jwt.verify(token, process.env.JWT_SECRET, (err, user) => {
    if (err) {
      return res.status(403).json({ message: 'Invalid token' });
    }
    req.user = user;
    next();
  });
}

app.get('/api/data', authenticateToken, (req, res) => {
  res.json({ data: 'Protected data' });
});
```

### API Key
```javascript
function authenticateApiKey(req, res, next) {
  const apiKey = req.headers['x-api-key'];
  
  if (!apiKey) {
    return res.status(401).json({ message: 'API key required' });
  }
  
  if (apiKey !== process.env.API_KEY) {
    return res.status(403).json({ message: 'Invalid API key' });
  }
  
  next();
}

app.get('/api/public-data', authenticateApiKey, (req, res) => {
  res.json({ data: 'Data' });
});
```

## Conditional Protection

### Optional Authentication
```javascript
function optionalAuth(req, res, next) {
  // Populate req.user if authenticated, but don't require it
  if (req.isAuthenticated()) {
    return next();
  }
  req.user = null;
  next();
}

app.get('/posts', optionalAuth, async (req, res) => {
  const posts = await Post.find();
  
  // Show different content based on auth status
  if (req.user) {
    // Include user-specific data
    res.json({ posts, favorites: req.user.favorites });
  } else {
    res.json({ posts });
  }
});
```

### Different Access Levels
```javascript
app.get('/posts/:id', optionalAuth, async (req, res) => {
  const post = await Post.findById(req.params.id);
  
  if (post.isPrivate && (!req.user || post.userId !== req.user.id)) {
    return res.status(403).json({ message: 'Private post' });
  }
  
  res.json({ post });
});
```

## Error Handling

### Centralized Error Handler
```javascript
function authErrorHandler(err, req, res, next) {
  if (err.name === 'UnauthorizedError') {
    return res.status(401).json({ 
      message: 'Invalid token',
      redirect: '/login'
    });
  }
  next(err);
}

app.use(authErrorHandler);
```

### Custom Error Class
```javascript
class AuthenticationError extends Error {
  constructor(message) {
    super(message);
    this.name = 'AuthenticationError';
    this.statusCode = 401;
  }
}

class AuthorizationError extends Error {
  constructor(message) {
    super(message);
    this.name = 'AuthorizationError';
    this.statusCode = 403;
  }
}

function ensureAuthenticated(req, res, next) {
  if (!req.isAuthenticated()) {
    throw new AuthenticationError('Authentication required');
  }
  next();
}

app.use((err, req, res, next) => {
  if (err instanceof AuthenticationError || err instanceof AuthorizationError) {
    return res.status(err.statusCode).json({ message: err.message });
  }
  next(err);
});
```

## Complete Example

```javascript
const express = require('express');
const app = express();

// Middleware
function ensureAuthenticated(req, res, next) {
  if (req.isAuthenticated()) return next();
  res.status(401).json({ message: 'Authentication required' });
}

function requireRole(role) {
  return (req, res, next) => {
    if (!req.isAuthenticated()) {
      return res.status(401).json({ message: 'Authentication required' });
    }
    if (req.user.role !== role) {
      return res.status(403).json({ message: 'Insufficient permissions' });
    }
    next();
  };
}

async function requireOwnership(req, res, next) {
  const post = await Post.findById(req.params.id);
  if (!post) {
    return res.status(404).json({ message: 'Not found' });
  }
  if (post.userId.toString() !== req.user.id.toString()) {
    return res.status(403).json({ message: 'Not authorized' });
  }
  req.post = post;
  next();
}

// Public routes
app.get('/', (req, res) => res.send('Home'));
app.post('/login', passport.authenticate('local'), (req, res) => {
  res.json({ user: req.user });
});

// Protected routes
app.get('/profile', ensureAuthenticated, (req, res) => {
  res.json({ user: req.user });
});

// Admin only
app.get('/admin', requireRole('admin'), (req, res) => {
  res.send('Admin panel');
});

// Owner only
app.put('/posts/:id', ensureAuthenticated, requireOwnership, (req, res) => {
  // Update req.post
  res.json({ message: 'Updated' });
});

app.listen(3000);
```

## Best Practices

1. **Always check authentication first** before authorization
2. **Use middleware** for reusable protection logic
3. **Return appropriate status codes** (401 for auth, 403 for authorization)
4. **Don't reveal sensitive info** in error messages
5. **Combine middleware** for complex requirements
6. **Apply at router level** for multiple routes
7. **Use role-based access** for different user types
8. **Check resource ownership** for user-specific data
9. **Handle errors gracefully** with proper messages
10. **Test thoroughly** with different user roles

## Key Takeaways

- **401 Unauthorized** = Not authenticated (need to log in)
- **403 Forbidden** = Authenticated but not authorized (insufficient permissions)
- **Middleware** = Reusable protection logic
- **req.isAuthenticated()** = Check if user is logged in
- **req.user** = Access authenticated user data
- **Role-based access** = Different permissions for different users
- **Resource ownership** = Users can only modify their own data
- **Combine middleware** = Stack multiple checks together
